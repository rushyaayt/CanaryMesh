GitHub Actions Workflow Example

Add this step to your .github/workflows/ci.yml to seed a canary token in every pipeline execution: 

name: Secure CI Pipeline
on: [push, pull_request]

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Code
        uses: actions/checkout@v4

      - name: Seed Ephemeral Honeytoken (CanaryMesh)
        run: |
          RESPONSE=$(curl -s -X POST "${{ secrets.CANARYMESH_URL }}/api/v1/seed/ci" \
            -H "Content-Type: application/json" \
            -d '{
              "repository": "${{ github.repository }}",
              "workflow": "${{ github.workflow }}",
              "run_id": "${{ github.run_id }}",
              "commit_sha": "${{ github.sha }}",
              "ttl_minutes": 90
            }')
          
          # Ingest token as decoy build secret
          KEY=$(echo "$RESPONSE" | grep -o '"token_value":"[^"]*' | cut -d'"' -f4)
          echo "DEPLOY_API_KEY=$KEY" >> "$GITHUB_ENV"
          echo "Seeded CanaryMesh ephemeral honeytoken into environment."

      - name: Run Build & Test Suite
        run: |
          npm test
          # If any rogue dependency or PR exfiltrates $DEPLOY_API_KEY and tests it,
          # CanaryMesh immediately triggers an incident alert!
